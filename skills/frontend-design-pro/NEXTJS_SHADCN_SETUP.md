# Next.js 16 + shadcn/ui Setup Guide

Step-by-step integration of Next.js 16 with shadcn/ui.

## 1. Create Next.js 16 Project

```bash
npx create-next-app@latest myapp \
  --typescript \
  --tailwind \
  --app \
  --eslint \
  --src-dir \
  --import-alias "@/*"

cd myapp
```

## 2. Initialize shadcn/ui

```bash
npx shadcn-ui@latest init
```

**Prompts:**
- Would you like to use TypeScript? → Yes
- Which style would you like to use? → Default (New York)
- Which color would you like as the base color? → Slate

**Creates:**
- `src/components/ui/` → UI components
- `src/lib/utils.ts` → cn() utility
- Updates `tsconfig.json` for path aliases

## 4. Update Layout for Providers

  description: "Generated by create next app",

  children,
}: {
  children: React.ReactNode
}) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>
        {children}
        <Toaster />  {/* Toast notifications */}
      </body>
    </html>
  )

## 7. Form Integration with react-hook-form

```bash
npm install react-hook-form
npx shadcn-ui@latest add form
```

```tsx
// src/components/LoginForm.tsx
"use client"

import { useForm } from "react-hook-form"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import {
  Form,
  FormControl,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from "@/components/ui/form"

export function LoginForm() {
  const form = useForm({
    defaultValues: {
      email: "",
      password: "",
    },
  })

  async function onSubmit(values) {
    const response = await fetch("/api/auth/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(values),
    })
    
    if (response.ok) {
      // Handle success
    }
  }

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-8">
        <FormField
          control={form.control}
          name="email"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Email</FormLabel>
              <FormControl>
                <Input type="email" placeholder="name@example.com" {...field} />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />
        
        <FormField
          control={form.control}
          name="password"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Password</FormLabel>
              <FormControl>
                <Input type="password" placeholder="••••••••" {...field} />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />
        
        <Button type="submit" className="w-full">
          Sign In
        </Button>
      </form>
    </Form>
  )
}
```

## 8. Dark Mode Support (Built-in)

shadcn/ui supports dark mode out of the box with Tailwind CSS.

```tsx
// src/app/layout.tsx
export default function RootLayout({ children }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body className="dark">{children}</body>
    </html>
  )
}
```

To toggle dark mode dynamically:

```tsx
// src/components/ThemeToggle.tsx
"use client"

import { Moon, Sun } from "lucide-react"
import { Button } from "@/components/ui/button"
import { useEffect, useState } from "react"

export function ThemeToggle() {
  const [theme, setTheme] = useState("light")
  
  useEffect(() => {
    const root = document.documentElement
    if (theme === "dark") {
      root.classList.add("dark")
      localStorage.setItem("theme", "dark")
    } else {
      root.classList.remove("dark")
      localStorage.setItem("theme", "light")
    }
  }, [theme])
  
  useEffect(() => {
    const saved = localStorage.getItem("theme") || "light"
    setTheme(saved)
  }, [])

  return (
    <Button
      variant="outline"
      size="icon"
      onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
    >
      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
    </Button>
  )
}
```

## Common Issues & Solutions

### Issue: Components not rendering

**Solution:** Make sure to use `"use client"` for components with hooks.

```tsx
"use client"  // Add this at top of file

import { useState } from "react"

export function MyComponent() {
  const [count, setCount] = useState(0)
  // ...
}
```

### Issue: Styling not applied

**Solution:** Check Tailwind is initialized and `globals.css` imported in layout.

```tsx
// src/app/layout.tsx
import "@/app/globals.css"  // Must import
```

### Issue: Types not recognized

**Solution:** Make sure `tsconfig.json` has correct path aliases.

```json
{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": {
      "@/*": ["./src/*"]
    }
  }
}
```

## Performance Tips

✅ **Use Server Components by Default**
- Server components don't ship JavaScript
- Fetch data directly in server components
- Only use "use client" when needed

✅ **Optimize Images**
```tsx
import Image from "next/image"

<Image
  src="/image.jpg"
  alt="Description"
  width={1200}
  height={600}
  priority  // For LCP images
/>
```

✅ **Dynamic Imports for Heavy Components**
```tsx
import dynamic from "next/dynamic"

const Modal = dynamic(() => import("@/components/Modal"), {
  loading: () => <div>Loading...</div>
})
```
